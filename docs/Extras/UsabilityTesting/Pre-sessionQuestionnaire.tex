\begin{enumerate}
    \item What is your name?
    \item How old are you?
    \item What gender do you identify with? 
    \item What is your occupation?

    \begin{enumerate}
        \item If \textbf{YES}, was it in the past 5 years?
        \item If \textbf{YES}, have you had physiotherapy after your knee surgery?
        \item If \textbf{NO}, have you had another kind of orthopaedic surgery?
    \end{enumerate}
\end{enumerate}
